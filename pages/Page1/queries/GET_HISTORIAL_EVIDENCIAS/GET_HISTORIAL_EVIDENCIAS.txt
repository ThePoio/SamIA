SELECT 
    e.id AS evidencia_id,
    t.titulo AS tarea_titulo,
    e.nombre_archivo,
    e.estado AS estado_evidencia,
    e.fecha_carga,
    h.dictamen,
    h.feedback_detallado,
    h.evaluado_en
FROM evidencias e
JOIN tareas t ON e.tarea_id = t.id
JOIN usuarios u ON e.usuario_id = u.id
LEFT JOIN historial_revisiones h ON h.evidencia_id = e.id
WHERE u.email = '{{ appsmith.user.email }}'
ORDER BY e.fecha_carga DESC;
