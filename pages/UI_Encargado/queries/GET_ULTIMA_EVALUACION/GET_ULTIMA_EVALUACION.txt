SELECT 
    hr.id AS revision_id,
    t.titulo AS tarea_titulo,
    COALESCE(t.seccion, 'General') AS seccion,
    e.nombre_archivo,
    e.fecha_carga,
    hr.dictamen,
    hr.feedback_detallado,
    hr.criterios_evaluados,
    hr.evaluado_en,
    (hr.tokens_prompt + hr.tokens_completion) AS total_tokens
FROM historial_revisiones hr
JOIN evidencias e ON hr.evidencia_id = e.id
JOIN tareas t ON e.tarea_id = t.id
JOIN usuarios u ON e.usuario_id = u.id
WHERE (
    u.email = '{{ appsmith.user.email }}' 
    OR '{{ appsmith.user.email }}' = '' 
    OR u.email IS NOT NULL
)
ORDER BY hr.evaluado_en DESC
LIMIT 1;
