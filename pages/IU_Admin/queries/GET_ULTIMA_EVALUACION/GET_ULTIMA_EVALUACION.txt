SELECT
    r.id AS revision_id,
    a.codigo_actividad,
    a.descripcion_actividad,
    e.nombre_archivo,
    e.fecha_carga,
    e.estado_revision,
CASE (r.coherencia_actividad_vs_recomendacion->>'cumple')
    WHEN 'true'  THEN 'CUMPLE'
    WHEN 'false' THEN 'NO CUMPLE'
    ELSE 'SIN DATO'
END AS res_actividad,
r.coherencia_actividad_vs_recomendacion->>'observacion' AS obs_actividad,

CASE (r.coherencia_evidencia_vs_actividad->>'cumple')
    WHEN 'true'  THEN 'CUMPLE'
    WHEN 'false' THEN 'NO CUMPLE'
    ELSE 'SIN DATO'
END AS res_evidencia,
r.coherencia_evidencia_vs_actividad->>'observacion' AS obs_evidencia,

CASE (r.coherencia_formato_entregado->>'cumple')
    WHEN 'true'  THEN 'CUMPLE'
    WHEN 'false' THEN 'NO CUMPLE'
    ELSE 'SIN DATO'
END AS res_formato,
r.coherencia_formato_entregado->>'observacion' AS obs_formato,
    r.alertas_inconsistencia,
    r.sugerencia_asistente,
    r.tokens_usados,
    r.procesado_en,
    d.dictamen,
    d.comentarios_auditor,
    d.dictaminado_en
FROM revisiones_asistente_ia r
JOIN evidencias_documentales e ON e.id = r.evidencia_id
JOIN actividades_comprometidas a ON a.id = e.actividad_id
JOIN usuarios u ON u.id = e.usuario_id
LEFT JOIN LATERAL (
    SELECT dictamen, comentarios_auditor, dictaminado_en
    FROM dictamenes_finales
    WHERE evidencia_id = e.id
    ORDER BY dictaminado_en DESC
    LIMIT 1
) d ON TRUE
WHERE u.email = {{ appsmith.user.email }}
ORDER BY r.procesado_en DESC
LIMIT 1;
