SELECT
    e.id AS evidencia_id,
    a.codigo_actividad,
    a.descripcion_actividad,
    e.nombre_archivo,
    e.estado_revision AS estado_evidencia,
    e.fecha_carga,
    d.dictamen,
    d.comentarios_auditor,
    d.dictaminado_en
FROM evidencias_documentales e
JOIN actividades_comprometidas a ON a.id = e.actividad_id
JOIN usuarios u ON u.id = e.usuario_id
LEFT JOIN dictamenes_finales d ON d.evidencia_id = e.id
WHERE u.email = {{ appsmith.user.email }}
ORDER BY e.fecha_carga DESC, d.dictaminado_en DESC;
