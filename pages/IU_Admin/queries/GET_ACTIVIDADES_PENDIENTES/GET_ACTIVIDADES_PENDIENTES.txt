SELECT
    actividad_id,
    codigo_actividad,
    descripcion_actividad,
    codigo_recomendacion,
    formato_solicitado,
    fecha_fin_programada,
    semaforo_estado
FROM vista_seguimiento_global
WHERE responsable_email = {{ appsmith.user.email }}
  AND porcentaje_avance < 100
ORDER BY fecha_fin_programada ASC;
