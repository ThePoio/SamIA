SELECT
    r.id AS triaje_id,
    r.sugerencia_asistente,
    CASE (r.coherencia_actividad_vs_recomendacion->>'cumple')
        WHEN 'true'  THEN 'CUMPLE'
        WHEN 'false' THEN 'NO CUMPLE'
        ELSE 'SIN DATO'
    END AS res_actividad,
    r.coherencia_actividad_vs_recomendacion->>'observacion' AS obs_actividad,
    CASE (r.coherencia_evidencia_vs_actividad->>'cumple')
        WHEN 'true'  THEN 'CUMPLE'
        WHEN 'false' THEN 'NO CUMPLE'
        ELSE 'SIN DATO'
    END AS res_evidencia,
    r.coherencia_evidencia_vs_actividad->>'observacion' AS obs_evidencia,
    CASE (r.coherencia_formato_entregado->>'cumple')
        WHEN 'true'  THEN 'CUMPLE'
        WHEN 'false' THEN 'NO CUMPLE'
        ELSE 'SIN DATO'
    END AS res_formato,
    r.coherencia_formato_entregado->>'observacion' AS obs_formato,
    r.alertas_inconsistencia,
    r.procesado_en
FROM revisiones_asistente_ia r
WHERE r.evidencia_id = {{ Select_Evidencia_Pendiente.selectedOptionValue || null }}::uuid
ORDER BY r.procesado_en DESC
LIMIT 1;
