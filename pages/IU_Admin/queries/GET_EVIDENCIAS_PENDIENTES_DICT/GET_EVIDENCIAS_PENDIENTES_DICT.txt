SELECT 
    e.id AS evidencia_id,
    e.nombre_archivo,
    a.codigo_actividad,
    a.descripcion_actividad,
    u.nombre AS responsable_nombre,
    r.sugerencia_asistente,
    e.fecha_carga
FROM evidencias_documentales e
JOIN actividades_comprometidas a ON e.actividad_id = a.id
JOIN usuarios u ON e.usuario_id = u.id
JOIN revisiones_asistente_ia r ON e.id = r.evidencia_id
WHERE e.estado_revision = 'ANALIZADO_IA'
ORDER BY e.fecha_carga ASC;
