INSERT INTO evidencias_documentales (
    actividad_id,
    usuario_id,
    nombre_archivo,
    ruta_almacenamiento,
    tipo_mime,
    estado_revision
)
SELECT
    {{ Select_Actividad.selectedOptionValue || null }}::uuid,
    u.id,
    {{ FilePicker_Evidencia.files[0].name }}::text,
    '/storage/evidencias/' || gen_random_uuid()::text || '_'
        || regexp_replace({{ FilePicker_Evidencia.files[0].name }}::text, '[^A-Za-z0-9._-]', '_', 'g'),
    {{ FilePicker_Evidencia.files[0].type }}::text,
    'PENDIENTE'
FROM usuarios u
WHERE u.email = {{ appsmith.user.email }}
  AND u.activo
  AND EXISTS (
      SELECT 1 FROM actividades_comprometidas a
      WHERE a.id = {{ Select_Actividad.selectedOptionValue || null }}::uuid
        AND puede_ver_area({{ appsmith.user.email }}::text, a.area_responsable_id)
  )
RETURNING id AS evidencia_id, actividad_id;
