SELECT
    r.id AS triaje_id,
    e.nombre_archivo,
    a.codigo_actividad,
    a.descripcion_actividad,
    r.coherencia_actividad_vs_recomendacion,
    r.coherencia_evidencia_vs_actividad,
    r.coherencia_formato_entregado,
    r.alertas_inconsistencia,
    r.sugerencia_asistente,
    r.procesado_en
FROM revisiones_asistente_ia r
JOIN evidencias_documentales e ON e.id = r.evidencia_id
JOIN actividades_comprometidas a ON a.id = e.actividad_id
JOIN usuarios u ON u.id = e.usuario_id
WHERE u.email = {{ appsmith.user.email }}
ORDER BY r.procesado_en DESC
LIMIT 1;
